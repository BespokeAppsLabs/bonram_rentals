## Connection Flow

### 1. Initialization

```typescript
// src/lib/convex-provider.tsx
import { ConvexProvider, ConvexReactClient } from "convex/react";

const convex = new ConvexReactClient(process.env.NEXT_PUBLIC_CONVEX_URL!);

export function ConvexClientProvider({ children }) {
  return <ConvexProvider client={convex}>{children}</ConvexProvider>;
}
```

### 2. Using in Layout

```typescript
// src/app/layout.tsx
import { ConvexClientProvider } from "@/lib/convex-provider";

export default function RootLayout({ children }) {
  return (
    <html>
      <body>
        <ConvexClientProvider>
          {children}
        </ConvexClientProvider>
      </body>
    </html>
  );
}
```

### 3. Querying Data

```typescript
// src/hooks/use-products.ts
import { useQuery } from "convex/react";
import { api } from "@/convex/_generated/api";

export function useProducts() {
  const products = useQuery(api.products.getAll);
  return { products, isLoading: products === undefined };
}
```

### 4. Using in Components

```typescript
// In any component
"use client";
import { useProducts } from "@/hooks/use-products";

export function ProductList() {
  const { products, isLoading } = useProducts();
  
  if (isLoading) return <div>Loading...</div>;
  
  return (
    <ul>
      {products?.map(product => (
        <li key={product._id}>{product.name}</li>
      ))}
    </ul>
  );
}
```

## API Functions

### Products API

| Function | Type | Description |
|----------|------|-------------|
| `products.getAll` | Query | Get all active products |
| `products.getByCategory` | Query | Filter by category |
| `products.getById` | Query | Get single product |
| `products.getByGuestCount` | Query | Products for guest count |
| `products.getCategories` | Query | List all categories |
| `products.create` | Mutation | Create product (Admin) |
| `products.update` | Mutation | Update product (Admin) |
| `products.remove` | Mutation | Soft delete product |

### Quotations API

| Function | Type | Description |
|----------|------|-------------|
| `quotations.getAll` | Query | All quotations (Admin) |
| `quotations.getByStatus` | Query | Filter by status |
| `quotations.getById` | Query | Single quote with items |
| `quotations.create` | Mutation | Create draft quote |
| `quotations.addItem` | Mutation | Add item to quote |
| `quotations.removeItem` | Mutation | Remove item |
| `quotations.updateItemQuantity` | Mutation | Change quantity |
| `quotations.updateStatus` | Mutation | Update status |
| `quotations.submitForReview` | Mutation | Submit for review |
| `quotations.updatePricing` | Mutation | Admin pricing update |

### Recommendations API

| Function | Type | Description |
|----------|------|-------------|
| `recommendations.getRecommendations` | Query | Smart suggestions |
| `recommendations.checkAvailability` | Query | Check stock |
| `recommendations.getProductsWithAvailability` | Query | Products with availability |

## Real-time Updates

Convex provides real-time updates automatically. When data changes:

1. Convex pushes updates via WebSocket
2. All subscribed components re-render
3. No manual refresh needed

Example:
```typescript
// This automatically updates when products change
const products = useQuery(api.products.getAll);
```

## Type Safety

Convex provides end-to-end type safety:

1. Schema defines types in `schema.ts`
2. Convex generates TypeScript types in `_generated/`
3. Hooks use generated types automatically

```typescript
// Type-safe product ID
import { Id } from "@/convex/_generated/dataModel";

const productId: Id<"products"> = "k57f8d9j2...";

// Type-safe function calls
const product = useQuery(api.products.getById, { id: productId });
//    ^? Product | null | undefined
```

## Error Handling

```typescript
export function useProducts() {
  const products = useQuery(api.products.getAll);
  
  return {
    products,
    isLoading: products === undefined,
    error: null, // Convex handles errors internally
  };
}
```

For mutations:
```typescript
const createProduct = useMutation(api.products.create);

async function handleCreate(data: ProductData) {
  try {
    const id = await createProduct(data);
    console.log("Created:", id);
  } catch (error) {
    console.error("Failed:", error);
  }
}
```
